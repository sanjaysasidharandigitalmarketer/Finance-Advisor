-- 1. EXTENSIONS & ENUMS CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; CREATE TYPE user_account_type AS ENUM ('personal', 'business', 'both'); CREATE TYPE transaction_type AS ENUM ('income', 'expense', 'transfer', 'loan_repayment', 'investment'); CREATE TYPE payment_mode AS ENUM ('upi', 'imps', 'neft', 'rtgs', 'debit_card', 'credit_card', 'cash', 'cheque'); CREATE TYPE loan_type AS ENUM ('home_loan', 'personal_loan', 'vehicle_loan', 'gold_loan', 'education_loan', 'business_loan', 'credit_card'); CREATE TYPE account_classification AS ENUM ('asset', 'liability', 'equity', 'revenue', 'expense'); -- 2. USER PROFILES & CONSENTS (DPDP ACT COMPLIANCE) CREATE TABLE profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, full_name TEXT NOT NULL, phone_number TEXT, account_type user_account_type NOT NULL DEFAULT 'personal', current_city TEXT, current_state TEXT, onboarding_completed BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE TABLE user_consents ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, terms_version TEXT NOT NULL, privacy_version TEXT NOT NULL, declaration_accepted BOOLEAN NOT NULL, ip_address TEXT, consented_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 3. BANK ACCOUNTS & CREDIT CARDS CREATE TABLE bank_accounts ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, bank_name TEXT NOT NULL, account_type TEXT NOT NULL, -- 'savings', 'current', 'salary' purpose TEXT NOT NULL, -- 'salary_credit', 'household_expense', 'savings', 'business' account_last_four VARCHAR(4), current_balance NUMERIC(15,2) NOT NULL DEFAULT 0.00, is_active BOOLEAN NOT NULL DEFAULT TRUE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE TABLE credit_cards ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, bank_name TEXT NOT NULL, card_name TEXT NOT NULL, credit_limit NUMERIC(15,2) NOT NULL, current_outstanding NUMERIC(15,2) NOT NULL DEFAULT 0.00, statement_date SMALLINT NOT NULL CHECK (statement_date BETWEEN 1 AND 31), due_date SMALLINT NOT NULL CHECK (due_date BETWEEN 1 AND 31), interest_rate_apr NUMERIC(5,2) NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 4. PERSONAL TRANSACTIONS & EXPENSES CREATE TABLE transactions ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, account_id UUID REFERENCES bank_accounts(id) ON DELETE SET NULL, credit_card_id UUID REFERENCES credit_cards(id) ON DELETE SET NULL, amount NUMERIC(15,2) NOT NULL CHECK (amount > 0), type transaction_type NOT NULL, payment_mode payment_mode NOT NULL, category TEXT NOT NULL, description TEXT, transaction_date DATE NOT NULL DEFAULT CURRENT_DATE, is_reconciled BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 5. LOANS & LIABILITIES CREATE TABLE loans ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, lender_name TEXT NOT NULL, loan_type loan_type NOT NULL, original_principal NUMERIC(15,2) NOT NULL, outstanding_principal NUMERIC(15,2) NOT NULL, interest_rate_pa NUMERIC(5,2) NOT NULL, monthly_emi NUMERIC(15,2) NOT NULL, tenure_months_remaining INT NOT NULL, next_payment_date DATE NOT NULL, is_secured BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 6. DOUBLE-ENTRY BUSINESS ACCOUNTING SYSTEM CREATE TABLE businesses ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, legal_name TEXT NOT NULL, trade_name TEXT, gstin VARCHAR(15), structure TEXT NOT NULL, -- 'proprietorship', 'llp', 'pvt_ltd', etc. created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE TABLE chart_of_accounts ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), business_id UUID NOT NULL REFERENCES businesses(id) ON DELETE CASCADE, account_code VARCHAR(20) NOT NULL, account_name TEXT NOT NULL, classification account_classification NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE(business_id, account_code) ); CREATE TABLE journal_entries ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), business_id UUID NOT NULL REFERENCES businesses(id) ON DELETE CASCADE, entry_date DATE NOT NULL DEFAULT CURRENT_DATE, narration TEXT NOT NULL, reference_number TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE TABLE journal_entry_lines ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), journal_entry_id UUID NOT NULL REFERENCES journal_entries(id) ON DELETE CASCADE, account_id UUID NOT NULL REFERENCES chart_of_accounts(id) ON DELETE RESTRICT, debit_amount NUMERIC(15,2) NOT NULL DEFAULT 0.00 CHECK (debit_amount >= 0), credit_amount NUMERIC(15,2) NOT NULL DEFAULT 0.00 CHECK (credit_amount >= 0), CHECK ((debit_amount > 0 AND credit_amount = 0) OR (credit_amount > 0 AND debit_amount = 0)) ); -- 7. ROW LEVEL SECURITY (RLS) POLICIES ALTER TABLE profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE user_consents ENABLE ROW LEVEL SECURITY; ALTER TABLE bank_accounts ENABLE ROW LEVEL SECURITY; ALTER TABLE credit_cards ENABLE ROW LEVEL SECURITY; ALTER TABLE transactions ENABLE ROW LEVEL SECURITY; ALTER TABLE loans ENABLE ROW LEVEL SECURITY; ALTER TABLE businesses ENABLE ROW LEVEL SECURITY; ALTER TABLE chart_of_accounts ENABLE ROW LEVEL SECURITY; ALTER TABLE journal_entries ENABLE ROW LEVEL SECURITY; ALTER TABLE journal_entry_lines ENABLE ROW LEVEL SECURITY; -- Standard user isolation policies CREATE POLICY "Profiles access" ON profiles FOR ALL USING (auth.uid() = id); CREATE POLICY "Consents access" ON user_consents FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Bank accounts access" ON bank_accounts FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Credit cards access" ON credit_cards FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Transactions access" ON transactions FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Loans access" ON loans FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Businesses access" ON businesses FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Chart of accounts access" ON chart_of_accounts FOR ALL USING (business_id IN (SELECT id FROM businesses WHERE user_id = auth.uid())); CREATE POLICY "Journal entries access" ON journal_entries FOR ALL USING (business_id IN (SELECT id FROM businesses WHERE user_id = auth.uid())); CREATE POLICY "Journal entry lines access" ON journal_entry_lines FOR ALL USING (journal_entry_id IN ( SELECT je.id FROM journal_entries je JOIN businesses b ON je.business_id = b.id WHERE b.user_id = auth.uid() ));
